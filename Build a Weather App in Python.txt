def make_label(parent, text, size=12, color=WHITE,
               bold=False, bg=None):
    return tk.Label(
        parent,
        text=text,
        font=("Segoe UI", size, "bold" if bold else "normal"),
        fg=color,
        bg=bg or parent.cget("bg")
    )


def make_card(parent, title, value, column):
    card = tk.Frame(
        parent,
        bg=PANEL,
        padx=14,
        pady=14,
        highlightthickness=1,
        highlightbackground="#2B4B65"
    )
    card.grid(row=0, column=column, padx=7, pady=8, sticky="nsew")

    make_label(
        card, title, 10, MUTED, True, PANEL
    ).pack(anchor="w")

    value_label = make_label(
        card, value, 15, WHITE, True, PANEL
    )
    value_label.pack(anchor="w", pady=(8, 0))

    return value_label

# ---------------- ERROR HANDLING ----------------
def show_error(message):
    status_label.config(text="Could not update weather.")
    messagebox.showerror("Weather App", message)

# ---------------- GET WEATHER DATA ----------------
def fetch_weather(city):
    if not API_KEY or API_KEY == ""ADD your API KEY"":
        root.after(
            0,
            lambda: show_error(
                "Please add your OpenWeatherMap API key "
                "to the API_KEY variable in the code."
            )
        )
        return

    try:
        response = session.get(
            "https://api.openweathermap.org/data/2.5/weather",
            params={
                "q": city,
                "appid": API_KEY,
                "units": "metric"
            },
            timeout=15
        )

        if response.status_code == 404:
            raise ValueError(
                "City not found. Please check the spelling."
            )

        if response.status_code == 401:
            raise ValueError(
                "Invalid API key or the key is not activated yet."
            )

        response.raise_for_status()
        data = response.json()

        weather = data["weather"][0]
        main = data["main"]
        wind = data.get("wind", {})

        # Find the city's local time zone
        lat = data["coord"]["lat"]
        lon = data["coord"]["lon"]

        timezone_name = timezone_finder.timezone_at(
            lat=lat, lng=lon
        )

        if timezone_name:
            local_zone = pytz.timezone(timezone_name)
            local_time = datetime.now(local_zone)
            time_text = local_time.strftime(
                "%A, %d %b | %I:%M %p"
            )
        else:
            time_text = "Local time unavailable"

        visibility = data.get("visibility")

        result = {
            "city": (
                f'{data.get("name", city)}, '
                f'{data.get("sys", {}).get("country", "")}'
            ).strip(", "),
            "time": time_text,
            "temp": round(main["temp"]),
            "condition": weather.get("main", "Unknown"),
            "description": weather.get(
                "description", ""
            ).capitalize(),
            "feels": round(main.get("feels_like", main["temp"])),
            "wind": wind.get("speed", 0),
            "humidity": main.get("humidity", "--"),
            "pressure": main.get("pressure", "--"),
            "visibility": (
                f"{visibility / 1000:.1f} km"
                if visibility is not None else "-- km"
            )
        }

        root.after(0, lambda: update_weather(result))

    except ValueError as error:
        root.after(0, lambda msg=str(error): show_error(msg))

    except requests.exceptions.Timeout:
        root.after(
            0,
            lambda: show_error(
                "Request timed out. Please try again."
            )
        )

    except requests.exceptions.ConnectionError:
        root.after(
            0,
            lambda: show_error(
                "Internet connection unavailable."
            )
        )

    except requests.exceptions.RequestException:
        root.after(
            0,
            lambda: show_error(
                "Weather request failed. Please try again."
            )
        )

    except (KeyError, IndexError, TypeError):
        root.after(
            0,
            lambda: show_error(
                "Could not read weather data. Please try again."
            )
        )


# ---------------- SEARCH FUNCTION ----------------
def get_weather(event=None):
    city = textfield.get().strip()

    if not city:
        messagebox.showwarning(
            "Missing City", "Please enter a city name."
        )
        return

    search_button.config(state="disabled", text="Loading...")
    status_label.config(text=f"Getting weather for {city}...")

    def worker():
        try:
            fetch_weather(city)
        finally:
            root.after(
                0,
                lambda: search_button.config(
                    state="normal", text="Search"
                )
            )

    threading.Thread(target=worker, daemon=True).start()


search_button.config(command=get_weather)
textfield.bind("<Return>", get_weather)

root.mainloop()
